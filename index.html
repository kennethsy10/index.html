const SHEET_NAME = 'Sheet1'; function doGet(e) { // 1. Serves the HTML file when opened directly via Apps Script Web App if (!e || !e.parameter || !e.parameter.action) { return HtmlService .createHtmlOutputFromFile('index') .setTitle('AES Secret Message Feed') .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL); } // 2. Returns feed data when requested externally via fetch() if (e.parameter.action === 'getMessages') { const data = getMessages(); return ContentService .createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); } } function doPost(e) { try { const contents = JSON.parse(e.postData.contents); const result = saveMessage(contents); return ContentService .createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService .createTextOutput(JSON.stringify({ success: false, error: error.message })) .setMimeType(ContentService.MimeType.JSON); } } function setup() { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(SHEET_NAME); } if (sheet.getLastRow() === 0) { sheet.appendRow([ 'Timestamp', 'Username', 'Ciphertext', 'Plaintext', 'key' ]); } sheet.setFrozenRows(1); } function saveMessage(data) { if (!data) throw new Error('No message data supplied.'); const username = String(data.username || '').trim(); const ciphertext = String(data.ciphertext || '').trim(); const plaintext = String(data.plaintext || ''); const key = String(data.key || ''); if (!username) throw new Error('Username is required.'); if (!ciphertext) throw new Error('Ciphertext is required.'); if (!key) throw new Error('Secret key is required.'); const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { setup(); sheet = ss.getSheetByName(SHEET_NAME); } sheet.appendRow([new Date(), username, ciphertext, plaintext, key]); return { success: true, message: 'Message published successfully.' }; } function getMessages() { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName(SHEET_NAME); if (!sheet || sheet.getLastRow() < 2) return []; const values = sheet .getRange(2, 1, sheet.getLastRow() - 1, 5) .getValues(); return values .reverse() .map(function(row, index) { return { id: index, timestamp: row[0] ? Utilities.formatDate( new Date(row[0]), Session.getScriptTimeZone(), 'yyyy-MM-dd HH:mm:ss' ) : '', username: String(row[1] || ''), ciphertext: String(row[2] || '') }; }); }
